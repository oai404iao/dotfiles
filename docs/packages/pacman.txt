# Arch packages used by this repository; select sections for your machine.
# Package names are unpinned and this is not an automatic install script.
# "pacman" includes official repositories except for the marked archlinuxcn section.

# Common: bootstrap, managed shell/Git/SSH configuration, and source checks.
git                    # Chezmoi source, Git identity, Neovim plugin bootstrap.
chezmoi                # Render and apply the dotfiles.
age                    # Decrypt the versioned SSH/Git configuration.
rbw                    # Bitwarden references and optional rbw SSH agent.
openssh                # SSH client/server identities and ssh-keygen checks.
systemd                # User services, loginctl and busctl (normally in Arch base).
python                 # Chezmoi modifiers, helpers, and offline tests.
bash                   # Managed Bash and helper scripts.
zsh                    # Managed Zsh; both shells are needed by check-source.sh.
pinentry               # pinentry-curses selected by the rbw configuration.
glib2                  # gio for safe-rm and its offline test.
coreutils              # rm, realpath, stat, install, timeout, etc.
util-linux             # flock and rfkill in desktop helpers.
findutils              # Source validation and helper scripts.
grep                   # Shell scripts and checks.
sed                    # Shell scripts and checks.
gawk                   # awk used by helpers and checks.
less                   # Expected pager.

# Configured CLI applications; install those you actually use.
neovim                 # LazyVim configuration and Kitty scrollback integration.
tmux                   # Managed tmux configuration.
zip                    # Zsh c() archive function.
aria2                  # Optional Zsh d alias (aria2c).
bash-completion        # Optional Bash completion, sourced if present.
# Oh My Zsh, Powerlevel10k and its plugins are optional user-managed files,
# not dependencies that chezmoi installs as Arch packages.

# Pi / agent tooling (independent of graphical and server roles).
uv                     # Tavily CLI via uv tool; also custom Waybar longshot.
# pnpm 12 is supplied by AUR pnpm-bin in aur.txt; Node 24 and npm/npx 12.1.0
# are user-managed through pnpm, not tracked here as Arch packages.
# Pi npm extensions and the uv-installed Tavily CLI are managed separately.

xdg-desktop-portal     # Screen sharing and desktop portals.

libsecret              # Secret Service client.
seahorse               # Keyring management UI.
fcitx5                 # Input method and generated theme.
fcitx5-rime            # Rime input method.
fcitx5-gtk             # GTK input-method integration.
fcitx5-qt              # Qt input-method integration.
nautilus               # File manager with managed GTK styling.
btop                   # Configured system monitor.
satty                  # Screenshot editor used by the custom profile.
matugen                # DMS palettes or custom Waypaper theme pipeline.
networkmanager         # nmtui and DMS network backend.
bluez                  # Bluetooth service used by desktop shells.

qt6ct                  # DMS-generated Qt palette.
